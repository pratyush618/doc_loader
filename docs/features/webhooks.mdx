Webhooks provide real-time notifications when document conversion jobs complete, allowing you to build responsive applications that react immediately to processing results.

## Overview

When you submit a job with a webhook URL, the Document Converter will send an HTTP POST request to your endpoint when the job completes (successfully or with failure).

<Info>
  Webhooks are optional but recommended for production applications to provide better user experience.
</Info>

<CodeGroup>

```python Job Creation
    }
)
```

    ```javascript
    const express = require('express');
    const app = express();

    app.use(express.json());

      console.log(`Job ${job.job_id} completed with status: ${job.status}`);

        console.error('Job failed:', job.error_message);
        handleJobFailure(job.job_id, job.error_message);
      }
      
      res.status(200).send('OK');
    });

      // Your document processing logic here

    }

    function handleJobFailure(jobId, error) {
      // Your error handling logic here
      console.error(`Job ${jobId} failed: ${error}`);
    }

    app.listen(3000, () => {
      console.log('Webhook server running on port 3000');
    });
    ```

    ```python
    from flask import Flask, request, jsonify
    import requests
    import logging

    app = Flask(__name__)
    logging.basicConfig(level=logging.INFO)

    def webhook_handler():
        job = request.json
        
        logging.info(f"Job {job['job_id']} completed with status: {job['status']}")
        
        if job['status'] == 'completed':

            response = requests.get(job['result_url'])
            if response.status_code == 200:

            else:
                logging.error(f"Failed to download result for job {job['job_id']}")

            logging.error(f"Job failed: {job.get('error_message', 'Unknown error')}")
            handle_job_failure(job['job_id'], job.get('error_message'))
        
        return jsonify({'status': 'received'}), 200

    def process_document(job_id, content):
        """Process the converted document"""

    def handle_job_failure(job_id, error):
    ```

    ```python
            background_tasks.add_task(process_completed_job, payload)
        elif payload.status == 'failed':
            background_tasks.add_task(handle_job_failure, payload.job_id, payload.error_message)
    ```

## Webhook Security

<Warning>
  Always validate webhook requests to ensure they come from your Document Converter instance.
</Warning>

<Tabs>
  <Tab title="IP Allowlist">
    ```python
    from flask import Flask, request, abort
    import ipaddress

    app = Flask(__name__)

    # Allow only specific IP addresses
    ALLOWED_IPS = ['127.0.0.1', '10.0.0.0/8', '192.168.0.0/16']

    def is_allowed_ip(ip):
        for allowed in ALLOWED_IPS:
            if '/' in allowed:
                if ipaddress.ip_address(ip) in ipaddress.ip_network(allowed):
                    return True
            else:
                if ip == allowed:
                    return True
        return False

    @app.before_request
    def limit_remote_addr():
        if not is_allowed_ip(request.remote_addr):
            abort(403)  # Forbidden
    ```
  </Tab>
  
  <Tab title="HMAC Signature">
    ```python
    import hmac
    import hashlib
    from flask import Flask, request, abort

    app = Flask(__name__)

    WEBHOOK_SECRET = 'your-webhook-secret-key'

    def verify_signature(payload, signature):
        expected_signature = hmac.new(
            WEBHOOK_SECRET.encode('utf-8'),
            payload,
            hashlib.sha256
        ).hexdigest()
        
        return hmac.compare_digest(f'sha256={expected_signature}', signature)

    @app.route('/webhook/document-complete', methods=['POST'])
    def webhook_handler():
        signature = request.headers.get('X-Webhook-Signature')
        if not signature:
            abort(403)
        
        payload = request.get_data()
        if not verify_signature(payload, signature):
            abort(403)
        
        # Process webhook
        job = request.json
        # ... handle job
        
        return {'status': 'received'}
    ```
  </Tab>

## Error Handling

<AccordionGroup>
  <Accordion title="Webhook Delivery Failures">
    If webhook delivery fails, the system will retry up to 3 times with exponential backoff:
    
    - **First retry**: After 1 second
    - **Second retry**: After 4 seconds  
    - **Third retry**: After 16 seconds
    
    After all retries fail, the webhook is marked as failed and no further attempts are made.
  </Accordion>
  
  <Accordion title="Timeout Handling">
    Webhooks have a default timeout of 30 seconds. Your endpoint should respond quickly:
    
    ```python
    @app.route('/webhook/document-complete', methods=['POST'])
    def webhook_handler():
        job = request.json
        
        # Respond immediately
        response = {'status': 'received'}
        
        # Process in background
        background_tasks.add_task(process_job, job)
        
        return response
    ```
  </Accordion>
  
  <Accordion title="Malformed Payloads">
    Always validate incoming webhook payloads:
    
    ```python
    @app.route('/webhook/document-complete', methods=['POST'])
    def webhook_handler():
        try:
            job = request.json
            
            # Validate required fields
            required_fields = ['job_id', 'status', 'progress']
            for field in required_fields:
                if field not in job:
                    return {'error': f'Missing field: {field}'}, 400
            
            # Process valid payload
            process_job(job)
            
        except Exception as e:
            logging.error(f"Webhook processing error: {e}")
            return {'error': 'Invalid payload'}, 400
        
        return {'status': 'received'}
    ```
  </Accordion>
</AccordionGroup>

## Best Practices

<CardGroup cols={2}>
  <Card title="Idempotency" icon="refresh">
    Make your webhook handler idempotent - it should handle duplicate deliveries gracefully by checking if the job was already processed.
  </Card>
  <Card title="Fast Response" icon="zap">
    Respond to webhooks quickly (< 5 seconds) to avoid timeouts. Process heavy operations in background tasks.
  </Card>
  <Card title="Proper Status Codes" icon="check-circle">
    Return appropriate HTTP status codes (200 for success, 4xx for client errors, 5xx for server errors).
  </Card>
  <Card title="Error Logging" icon="file-text">
    Log all webhook events and errors for debugging and monitoring purposes.
  </Card>
</CardGroup>

## Monitoring Webhooks

<CodeGroup>

```python Webhook Monitoring
import logging
import time
from collections import defaultdict

# Track webhook metrics
webhook_metrics = defaultdict(int)

@app.route('/webhook/document-complete', methods=['POST'])
def webhook_handler():
    start_time = time.time()
    
    try:
        job = request.json
        
        # Process webhook
        process_job(job)
        
        # Track success
        webhook_metrics['success'] += 1
        
    except Exception as e:
        webhook_metrics['error'] += 1
        logging.error(f"Webhook error: {e}")
        
    finally:
        # Track processing time
        processing_time = time.time() - start_time
        webhook_metrics['total_time'] += processing_time
        webhook_metrics['count'] += 1
        
        # Log metrics periodically
        if webhook_metrics['count'] % 100 == 0:
            avg_time = webhook_metrics['total_time'] / webhook_metrics['count']
            logging.info(f"Webhook metrics: {dict(webhook_metrics)}, avg_time: {avg_time:.2f}s")
    
    return {'status': 'received'}
```

```python Health Check Endpoint
@app.route('/webhook/health', methods=['GET'])
def webhook_health():
    return {
        'status': 'healthy',
        'metrics': dict(webhook_metrics),
        'timestamp': time.time()
    }
```

## Common Issues

<AccordionGroup>
  <Accordion title="Webhook Not Receiving Data">
    **Causes:**
    - Firewall blocking incoming requests
    - Incorrect URL or endpoint
    - SSL/TLS certificate issues
    - Server not running
    
    **Solutions:**
    - Test with curl or Postman
    - Check firewall rules
    - Verify SSL certificate
    - Use ngrok for local testing
  </Accordion>
  
  <Accordion title="Webhook Timeouts">
    **Causes:**
    - Slow processing in webhook handler
    - Database or external API calls
    - Large file downloads
    
    **Solutions:**
    - Process in background tasks
    - Return response immediately
    - Implement proper timeout handling
    - Use async processing
  </Accordion>
  
  <Accordion title="Duplicate Webhooks">
    **Causes:**
    - Network issues causing retries
    - Server errors during processing
    - Multiple webhook URLs configured
    
    **Solutions:**
    - Implement idempotency checks
    - Use unique job IDs
    - Store processed webhook IDs
    - Return proper HTTP status codes
  </Accordion>
</AccordionGroup>
